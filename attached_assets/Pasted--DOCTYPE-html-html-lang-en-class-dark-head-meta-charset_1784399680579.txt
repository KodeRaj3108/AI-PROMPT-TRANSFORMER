function showToast(message, tone = 'default') {
  const container = document.getElementById('toastContainer');
  const toneClasses = tone === 'error'
    ? 'border-rose-400/30 bg-rose-500/15 text-rose-100'
    : 'border-white/10 bg-white/10 text-slate-100';
  const el = document.createElement('div');
  el.className = `animate-toastIn glass px-4 py-2.5 rounded-xl text-xs font-medium shadow-lg ${toneClasses}`;
  el.textContent = message;
  container.appendChild(el);
  setTimeout(() => {
    el.style.transition = 'opacity 0.3s ease';
    el.style.opacity = '0';
    setTimeout(() => el.remove(), 300);
  }, 2400);
}

// =========================================================================
// CLIPBOARD
// =========================================================================
async function copyToClipboard(text) {
  try {
    await navigator.clipboard.writeText(text);
    return true;
  } catch (e) {
    try {
      const ta = document.createElement('textarea');
      ta.value = text;
      ta.style.position = 'fixed';
      ta.style.opacity = '0';
      document.body.appendChild(ta);
      ta.select();
      document.execCommand('copy');
      document.body.removeChild(ta);
      return true;
    } catch (e2) {
      return false;
    }
  }
}

// =========================================================================
// COPY / TEST HANDLERS
// =========================================================================
async function handleCopy(modelKey) {
  if (!currentOutputs || !currentOutputs[modelKey]) return;
  const ok = await copyToClipboard(currentOutputs[modelKey]);
  showToast(ok ? `${labelFor(modelKey)} prompt copied` : 'Copy failed — select and copy manually', ok ? 'default' : 'error');
}

async function handleTest(modelKey) {
  if (!currentOutputs || !currentOutputs[modelKey]) return;
  const profile = MODEL_PROFILES.find(p => p.key === modelKey);
  const text = currentOutputs[modelKey];

  if (modelKey === 'cursor') {
    cursorClipboardText = text;
    await copyToClipboard(text);
    openCursorModal();
    return;
  }

  const ok = await copyToClipboard(text);
  window.open(profile.testUrl, '_blank', 'noopener,noreferrer');
  showToast(ok
    ? `Copied — paste into ${profile.label} once the tab opens`
    : `Opened ${profile.label} — copy failed, select text manually`, ok ? 'default' : 'error');
}

function labelFor(key) {
  return MODEL_PROFILES.find(p => p.key === key)?.label || key;
}

// =========================================================================
// TRANSFORM — calls Claude via the sandbox's built-in API access
// =========================================================================
async function handleTransform() {
  if (isLoading) return;
  hideError();

  const roughInput = document.getElementById('roughInput').value.trim();
  if (!roughInput) {
    showError('Enter a rough prompt before transforming.');
    return;
  }

  isLoading = true;
  setButtonLoading(true);
  renderOutputCards(null, true);

  try {
    const outputs = await callClaudeForTransformation(roughInput);
    currentOutputs = outputs;
    renderOutputCards(outputs, false);
    saveToHistory(roughInput, outputs);
    renderHistory();
    showToast('Prompt transformed into 5 model-tuned versions');
  } catch (err) {
    console.error(err);
    showError(err.message || 'Something went wrong while transforming the prompt. Please try again.');
    renderOutputCards(null, false);
  } finally {
    isLoading = false;
    setButtonLoading(false);
  }
}

function setButtonLoading(loading) {
  const btn = document.getElementById('transformBtn');
  const label = document.getElementById('transformBtnLabel');
  btn.disabled = loading;
  label.textContent = loading ? 'Transforming…' : 'Transform Prompt';
}

async function callClaudeForTransformation(roughInput) {
  const systemPrompt = `You are a prompt engineering specialist. You will be given a single rough, unstructured prompt from a user. Your job is to rewrite it into 5 distinct, optimized versions — one tailored for each of these AI systems, each following that system's specific design conventions:

1. chatgpt: Clear step-by-step formatting, an explicit system-role definition line, and markdown structure (headers, numbered lists).
2. claude: Use XML tags to isolate variables, e.g. <context>, <instructions>, <input> — structurally clear, and leave room for deep step-by-step reasoning.
3. gemini: Organize with logical headings, give explicit formatting instructions, keep it token-efficient and tightly constrained.
4. grok: Direct, no-nonsense, high-impact brevity, and include real-time context scaffolding (phrase it assuming access to current/live information where relevant).
5. cursor: Precise code-context generation — reference file/folder structure where relevant, specify language/framework, and phrase it like a .cursorrules-style system directive.

Rewrite the user's underlying intent faithfully — do not add requirements they didn't imply. Keep each of the 5 outputs compact (roughly 3-7 lines / under 90 words each) since output length is limited, while still clearly reflecting that model's distinct conventions.

Respond with ONLY a raw JSON object, no markdown code fences, no preamble, no commentary, in exactly this shape:
{"chatgpt": "...", "claude": "...", "gemini": "...", "grok": "...", "cursor": "..."}`;

  const response = await fetch("https://api.anthropic.com/v1/messages", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "claude-sonnet-4-6",
      max_tokens: 1000,
      system: systemPrompt,
      messages: [
        { role: "user", content: `Rough prompt to transform:\n\n${roughInput}` }
      ],
    })
  });

  if (!response.ok) {
    throw new Error(`Request failed (status ${response.status}). Please try again.`);
  }

  const data = await response.json();
  const textBlocks = (data.content || []).filter(b => b.type === 'text').map(b => b.text);
  const rawText = textBlocks.join('\n').trim();

  if (!rawText) {
    throw new Error('The engine returned an empty response. Please try again.');
  }

  const cleaned = rawText.replace(/^```json/i, '').replace(/^```/, '').replace(/```$/, '').trim();

  let parsed;
  try {
    parsed = JSON.parse(cleaned);
  } catch (e) {
    throw new Error('Could not parse a structured response from the engine. Please try again.');
  }

  const required = ['chatgpt', 'claude', 'gemini', 'grok', 'cursor'];
  for (const key of required) {
    if (typeof parsed[key] !== 'string' || !parsed[key].trim()) {
      throw new Error(`The response was missing the "${key}" output. Please try again.`);
    }
  }

  return parsed;
}

// =========================================================================
// HISTORY (localStorage)
// =========================================================================
function loadHistory() {
  try {
    const raw = localStorage.getItem(HISTORY_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch (e) {
    return [];
  }
}

function saveToHistory(roughInput, outputs) {
  const history = loadHistory();
  const entry = {
    id: `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
    timestamp: Date.now(),
    rough: roughInput,
    outputs,
  };
  history.unshift(entry);
  const trimmed = history.slice(0, MAX_HISTORY);
  try {
    localStorage.setItem(HISTORY_KEY, JSON.stringify(trimmed));
  } catch (e) {
    // storage full or unavailable — fail silently, transformation still shown on screen
  }
}

function renderHistory() {
  const history = loadHistory();
  const list = document.getElementById('historyList');
  const empty = document.getElementById('historyEmpty');

  list.innerHTML = '';

  if (history.length === 0) {
    list.appendChild(empty);
    return;
  }

  history.forEach(entry => {
    const item = document.createElement('button');
    item.className = 'history-item text-left w-full rounded-xl px-3 py-2.5 bg-white/5 border border-white/10 hover:bg-white/10 hover:border-violet-400/30 transition group';
    item.onclick = () => loadHistoryEntry(entry.id);

    const date = new Date(entry.timestamp);
    const timeStr = date.toLocaleDateString(undefined, { month: 'short', day: 'numeric' }) + ' · ' + date.toLocaleTimeString(undefined, { hour: '2-digit', minute: '2-digit' });

    item.innerHTML = `
      <div class="flex items-start gap-2">
        <span class="history-dot h-1.5 w-1.5 rounded-full bg-violet-400 mt-1.5 transition-transform"></span>
        <div class="min-w-0 flex-1">
          <p class="text-xs text-slate-200 truncate">${escapeHTML(entry.rough)}</p>
          <p class="text-[10px] text-slate-500 font-mono mt-0.5">${timeStr}</p>
        </div>
      </div>
    `;
    list.appendChild(item);
  });
}

function loadHistoryEntry(id) {
  const history = loadHistory();
  const entry = history.find(h => h.id === id);
  if (!entry) return;

  document.getElementById('roughInput').value = entry.rough;
  currentOutputs = entry.outputs;
  renderOutputCards(entry.outputs, false);
  hideError();
  showToast('Loaded from history');
}

function clearHistory() {
  try {
    localStorage.removeItem(HISTORY_KEY);
  } catch (e) {}
  renderHistory();
  showToast('History cleared');
}
